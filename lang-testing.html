⚡ Jev Language Testing Lab

Modular, runnable call examples for Jev (TypeSafe System One) across every major language and runtime — from server-side Rust to browser-native JavaScript. Each tab is a self-contained implementation module you can drop into your own stack.

🔐
Server ↔ Client Architecture: All examples that carry an API key run server-side (Rust Actix-web, Python FastAPI, Node.js, Go net/http, Java Spring Boot) and expose a thin /api/judge proxy. Browser-side examples (TypeScript fetch) call the local proxy — never the TypeSafe API directly. This keeps TYPESAFE_API_KEY out of your JavaScript bundles. See the HTTP Raw tab for curl/REST examples that can be used in any environment.

🦀 Rust + Actix-web
Server-side

Overview

Call the TypeSafe Jev API from a Rust Actix-web server using the reqwest HTTP client and serde_json for JSON serialization. The server exposes a /api/judge endpoint that browsers and CLI clients call. The TypeSafe API key never leaves the server.

Setup Steps

1 Add dependencies to Cargo.toml: actix-web, reqwest, serde, serde_json, tokio
2 Set environment variable: export TYPESAFE_API_KEY=sk-ts-...
3 Run with: cargo run --release. Server starts on http://127.0.0.1:8090
4 POST to /api/judge with a JSON body containing text to classify

Jev Primitives Used

📊 Noul — billing? 🎯 Choice — department 📈 Score — urgency ⚡ Fan-out (3 parallel)

Primitive Reference

Primitive Return Use When
Noul float 0–1 Yes/no probability
Choice string key + probabilities Pick one of N options
Score float on ordered levels Degree on a spectrum
🔐 Key Security: The TYPESAFE_API_KEY is read once at startup via std::env::var and stored in AppState. It is never serialized into any HTTP response or logged. Clients only see the /api/judge proxy response.
src/main.rs 
// src/main.rs — JevTools Rust Example
// Actix-web server that proxies Jev (TypeSafe System One) calls
// API key stays server-side; clients POST to /api/judge

use actix_web::{get, post, web, App, HttpResponse, HttpServer, Responder};
use reqwest::Client;
use serde::{Deserialize, Serialize};
use serde_json::{json, Value};
use std::env;

// ─── Shared application state ───────────────────────────────────────
#[derive(Clone)]
struct AppState {
    api_key: String,
    http:    Client,
}

// ─── Request / Response types ────────────────────────────────────────
#[derive(Deserialize)]
struct JudgeRequest {
    text: String,
}

#[derive(Serialize)]
struct JudgeResponse {
    billing:    f64,           // Noul: 0–1 probability
    department: String,        // Choice: winning key
    confidence: f64,           // Choice confidence
    urgency:    f64,           // Score: 0–(levels-1)
    model:      String,
}

// ─── Health check ────────────────────────────────────────────────────
#[get("/api/status")]
async fn status() -> impl Responder {
    HttpResponse::Ok().json(json!({
        "engine": "jev-latest",
        "runtime": "rust/actix-web",
        "ok": true
    }))
}

// ─── Jev proxy endpoint ───────────────────────────────────────────────
#[post("/api/judge")]
async fn judge(
    data:    web::Data<AppState>,
    payload: web::Json<JudgeRequest>,
) -> impl Responder {

    // Build the TypeSafe Jev request body with 3 parallel questions
    let body = json!({
        "model": "jev-latest",
        "state": { "message": payload.text },
        "questions": {
            "billing": {
                "type": "noul",
                "instructions": "Is `message` about a billing or payment issue?"
            },
            "department": {
                "type": "choice",
                "instructions": "Which support team should handle `message`?",
                "criteria": {
                    "billing":   "Payment, subscription, or invoice issues.",
                    "technical": "Bugs, API errors, or integration problems.",
                    "sales":     "Pricing, upgrades, or new account questions.",
                    "other":     "None of the above."
                }
            },
            "urgency": {
                "type": "score",
                "instructions": "How urgent does `message` appear?",
                "criteria": [
                    "Can wait days — no expressed urgency.",
                    "Should be resolved this week — some concern.",
                    "Needs action today — strong urgency or anger."
                ]
            }
        }
    });

    // Call TypeSafe API
    let resp = data.http
        .post("https://api.typesafe.ai/v1/systemone")
        .bearer_auth(&data.api_key)
        .json(&body)
        .send()
        .await;

    let resp = match resp {
        Ok(r) => r,
        Err(e) => {
            return HttpResponse::BadGateway()
                .json(json!({ "error": e.to_string() }));
        }
    };

    let jev: Value = match resp.json().await {
        Ok(v) => v,
        Err(e) => {
            return HttpResponse::InternalServerError()
                .json(json!({ "error": e.to_string() }));
        }
    };

    // Extract typed answers from response
    let answers = &jev["answers"];

    let billing = answers["billing"]["noul"]
        .as_f64().unwrap_or(0.0);

    let department = answers["department"]["choice"]
        .as_str().unwrap_or("other").to_string();

    let confidence = answers["department"]["confidence"]
        .as_f64().unwrap_or(0.0);

    let urgency = answers["urgency"]["score"]
        .as_f64().unwrap_or(0.0);

    let model = jev["model"]
        .as_str().unwrap_or("jev-latest").to_string();

    // Confidence-gated routing in code (policy stays server-side)
    if confidence < 0.5 {
        println!("[jev] Low confidence {:.2} → flag for review", confidence);
    }

    HttpResponse::Ok().json(JudgeResponse {
        billing, department, confidence, urgency, model,
    })
}

// ─── Main ─────────────────────────────────────────────────────────────
#[actix_web::main]
async fn main() -> std::io::Result<()> {
    let api_key = env::var("TYPESAFE_API_KEY")
        .expect("TYPESAFE_API_KEY must be set");

    let state = web::Data::new(AppState {
        api_key,
        http: Client::new(),

    println!("🦀 JevTools Rust server → http://127.0.0.1:8090");

            .app_data(state.clone())

    .await
}

# jev_judge.py — Synchronous Jev SDK example
# pip install typesafe-sdk

from typesafe_sdk import Choice, Noul, Score, TypeSafeClient

def judge_message(text: str) -> dict:
    """Run 3 parallel Jev judgments on a support message."""
    with TypeSafeClient() as client:
        response = client.system_one(
            state={"message": text},
            questions={
                "billing": Noul(
                    instructions="Is `message` about a billing or payment issue?"
                ),
                "department": Choice(
                    instructions="Which support team should handle `message`?",
                    criteria={
                        "billing":   "Payment, subscription, or invoice issues.",
                        "technical": "Bugs, API errors, or integration problems.",
                        "sales":     "Pricing, upgrades, or new account questions.",
                        "other":     "None of the above.",
                    },
                ),
                "urgency": Score(
                    instructions="How urgent does `message` appear?",
                    criteria=[
                        "Can wait days — no expressed urgency.",
                        "Should be resolved this week.",
                        "Needs action today — strong urgency.",
                    ],
                ),
            },
        )

    # Typed shorthand accessors
    billing    = response.nouls["billing"].noul          # float 0–1
    department = response.choices["department"].choice    # str
    confidence = response.choices["department"].confidence # float 0–1
    urgency    = response.scores["urgency"].score         # float 0–2

    # Confidence-gated routing (policy in code, not questions)
    if confidence < 0.5:
        action = "route_to_human"
    elif department == "billing":
        action = "open_billing_ticket"
    elif department == "technical":
        action = "open_tech_ticket"
    else:
        action = "open_general_ticket"

    return {
        "billing":    billing,
        "department": department,
        "confidence": confidence,
        "urgency":    urgency,
        "action":     action,
    }


if __name__ == "__main__":
    result = judge_message("I was charged twice. Fix this ASAP!")
    for k, v in result.items():
        print(f"  {k}: {v}")

// jev_judge.ts — TypeScript SDK (Node.js server-side)
// npm install @typesafe-ai/sdk
// npx tsx jev_judge.ts

import { choice, noul, score, TypeSafeClient } from "@typesafe-ai/sdk";

const client = new TypeSafeClient();
// Reads TYPESAFE_API_KEY from process.env automatically

async function judgeMessage(text: string) {
  const response = await client.systemOne({
    state: { message: text },
    questions: {
      billing: noul("Is `message` about a billing or payment issue?"),

      department: choice("Which support team handles `message`?", {
        billing:   "Payment, subscription, or invoice issues.",
        technical: "Bugs, API errors, or integration problems.",
        sales:     "Pricing, upgrades, or new account questions.",
        other:     "None of the above.",
      }),

      urgency: score("How urgent does `message` appear?", [
        "Can wait days — no expressed urgency.",
        "Should be resolved this week — some concern.",
        "Needs action today — strong urgency or anger.",
      ]),
    },
  });

  // Answer types are fully inferred — no casting needed
  const billing    = response.answers.billing.noul;         // number
  const dept       = response.answers.department.choice;    // "billing"|"technical"|"sales"|"other"
  const confidence = response.answers.department.confidence; // number
  const urgency    = response.answers.urgency.score;         // number

  // Confidence-gated routing in code
  let action: string;
  if (confidence < 0.5) {
    action = "route_to_human";
  } else if (dept === "billing") {
    action = "open_billing_ticket";
  } else if (dept === "technical") {
    action = "open_tech_ticket";
  } else {
    action = "open_general_ticket";
  }

  return { billing, department: dept, confidence, urgency, action };
}

// main.go — JevTools Go example using net/http + encoding/json
package main

import (
    "bytes"
    "encoding/json"
    "fmt"
    "io"
    "log"
    "net/http"
    "os"
)

const jevEndpoint = "https://api.typesafe.ai/v1/systemone"

type JudgeRequest struct {
    Text string `json:"text"`
}

type JudgeResult struct {
    Billing    float64 `json:"billing"`
    Department string  `json:"department"`
    Confidence float64 `json:"confidence"`
    Urgency    float64 `json:"urgency"`
}

func callJev(apiKey, text string) (*JudgeResult, error) {
    body, _ := json.Marshal(map[string]any{
        "model": "jev-latest",
        "state": map[string]string{"message": text},
        "questions": map[string]any{
            "billing": map[string]string{
                "type":         "noul",
                "instructions": "Is `message` about billing?",
            },
            "department": map[string]any{
                "type":         "choice",
                "instructions": "Which team handles `message`?",
                "criteria": map[string]string{
                    "billing":   "Payment issues.",
                    "technical": "Bug reports.",
                    "other":     "All else.",
                },
            },
            "urgency": map[string]any{
                "type":         "score",
                "instructions": "How urgent is `message`?",
                "criteria":     []string{"Low", "Medium", "High"},
            },
        },
    })

    req, _ := http.NewRequest(http.MethodPost, jevEndpoint,
        bytes.NewReader(body))
    req.Header.Set("Authorization", "Bearer "+apiKey)
    req.Header.Set("Content-Type", "application/json")

    resp, err := http.DefaultClient.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    var raw map[string]any
    if err := json.NewDecoder(resp.Body).Decode(&raw); err != nil {
        return nil, err
    }

    answers := raw["answers"].(map[string]any)
    result := &JudgeResult{
        Billing:    answers["billing"].(map[string]any)["noul"].(float64),
        Department: answers["department"].(map[string]any)["choice"].(string),
        Confidence: answers["department"].(map[string]any)["confidence"].(float64),
        Urgency:    answers["urgency"].(map[string]any)["score"].(float64),
    }
    return result, nil
}

func judgeHandler(apiKey string) http.HandlerFunc {
    return func(w http.ResponseWriter, r *http.Request) {
        var req JudgeRequest
        json.NewDecoder(r.Body).Decode(&req)

        result, err := callJev(apiKey, req.Text)
        if err != nil {
            http.Error(w, err.Error(), http.StatusBadGateway)
            return
        }
        w.Header().Set("Content-Type", "application/json")
        json.NewEncoder(w).Encode(result)
    }
}

func main() {
    apiKey := os.Getenv("TYPESAFE_API_KEY")
    if apiKey == "" {
        log.Fatal("TYPESAFE_API_KEY not set")
    }
    http.HandleFunc("/api/judge", judgeHandler(apiKey))
    fmt.Println("🐹 Go/Jev server → http://127.0.0.1:8092")
    log.Fatal(http.ListenAndServe(":8092", nil))
}

// JevController.java — Spring Boot REST endpoint for Jev
package com.example.jevtools;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.web.bind.annotation.*;

import java.net.URI;
import java.net.http.*;
import java.util.Map;

@RestController
@RequestMapping("/api")
public class JevController {

    @Value("${typesafe.api-key}")
    private String apiKey;

    private final ObjectMapper mapper = new ObjectMapper();
    private final HttpClient   http   = HttpClient.newHttpClient();

    @PostMapping("/judge")
    public Map<String, Object> judge(@RequestBody Map<String, String> req)
            throws Exception {

        var body = mapper.writeValueAsString(Map.of(
            "model", "jev-latest",
            "state", Map.of("message", req.get("text")),
            "questions", Map.of(
                "billing", Map.of(
                    "type", "noul",
                    "instructions", "Is `message` about a billing issue?"
                ),
                "department", Map.of(
                    "type", "choice",
                    "instructions", "Which team handles `message`?",
                    "criteria", Map.of(
                        "billing", "Payment issues.",
                        "technical", "Bug reports.",
                        "other", "All else."
                    )
                ),
                "urgency", Map.of(
                    "type", "score",
                    "instructions", "How urgent is `message`?",
                    "criteria", new String[]{"Low", "Medium", "High"}
                )
            )
        ));

        var httpReq = HttpRequest.newBuilder()
            .uri(URI.create("https://api.typesafe.ai/v1/systemone"))
            .header("Authorization", "Bearer " + apiKey)
            .header("Content-Type", "application/json")
            .POST(HttpRequest.BodyPublishers.ofString(body))
            .build();

        var resp    = http.send(httpReq, HttpResponse.BodyHandlers.ofString());
        JsonNode jev = mapper.readTree(resp.body());
        JsonNode ans = jev.path("answers");

        return Map.of(
            "billing",    ans.path("billing")   .path("noul")      .asDouble(),
            "department", ans.path("department").path("choice")    .asText(),
            "confidence", ans.path("department").path("confidence").asDouble(),
            "urgency",    ans.path("urgency")   .path("score")     .asDouble()
        );
    }
}

        var resp = HttpClient.newHttpClient()

// Full request schema (JSON)
{
  "model": "jev-latest",          // required
  "state": "<string | object | array>",  // required
  "questions": {
    "<question_id>": {             // your code key; never sent to model
      "type": "noul | choice | score",
      "instructions": "<string | object | array>",
      "criteria": // required for choice/score; optional for noul
        // Choice:  { "key": "description", ... }
        // Score:   [ "level 0 desc", "level 1 desc", ... ]
        // Noul:    { "yes": "...", "no": "..." }  (optional)
    }
  }
}

// - state is shared across all questions
// - question_id is for your code only, not sent to Jev
